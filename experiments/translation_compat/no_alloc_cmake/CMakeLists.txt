cmake_minimum_required(VERSION 3.14)
project(hyv3_descriptor_check LANGUAGES CXX)
if(NOT LLAMA_SOURCE OR NOT LLAMA_BUILD)
    message(FATAL_ERROR "Set LLAMA_SOURCE and LLAMA_BUILD to exact patched source/build")
endif()
add_executable(no_alloc ../no_alloc.cpp)
target_compile_features(no_alloc PRIVATE cxx_std_17)
target_compile_definitions(no_alloc PRIVATE LLAMA_SHARED GGML_SHARED)
target_include_directories(no_alloc PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
target_link_libraries(no_alloc PRIVATE "${LLAMA_BUILD}/src/Release/llama.lib" "${LLAMA_BUILD}/ggml/src/Release/ggml.lib" "${LLAMA_BUILD}/ggml/src/Release/ggml-base.lib")
